CREATE   PROCEDURE inventar_SpcHidr.sp_rep_prod_sync_schema
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @schema SYSNAME = 'inventar_SpcHidr';
    DECLARE @src_table SYSNAME = 'Datos_conciliados_of';
    DECLARE @tgt1 SYSNAME = 'rep_prod_ajustado';
    DECLA
RE @tgt2 SYSNAME = 'rep_prod_ajustado_stg';

    DECLARE @missing TABLE
    (
        target_table SYSNAME,
        column_name SYSNAME,
        data_type SYSNAME,
        char_len INT NULL,
        num_prec INT NULL,
        num_scale INT NULL,
        d
t_prec INT NULL
    );

    -- missing para rep_prod_ajustado
    INSERT INTO @missing (target_table, column_name, data_type, char_len, num_prec, num_scale, dt_prec)
    SELECT
        @tgt1,
        s.COLUMN_NAME,
        s.DATA_TYPE,
        s.CHARACTER
_MAXIMUM_LENGTH,
        s.NUMERIC_PRECISION,
        s.NUMERIC_SCALE,
        s.DATETIME_PRECISION
    FROM INFORMATION_SCHEMA.COLUMNS s
    LEFT JOIN INFORMATION_SCHEMA.COLUMNS t
        ON  t.TABLE_SCHEMA = s.TABLE_SCHEMA
        AND t.TABLE_NAME   = @
tgt1
        AND t.COLUMN_NAME  = s.COLUMN_NAME
    WHERE s.TABLE_SCHEMA = @schema
      AND s.TABLE_NAME   = @src_table
      AND t.COLUMN_NAME IS NULL;

    -- missing para rep_prod_ajustado_stg
    INSERT INTO @missing (target_table, column_name, data_
type, char_len, num_prec, num_scale, dt_prec)
    SELECT
        @tgt2,
        s.COLUMN_NAME,
        s.DATA_TYPE,
        s.CHARACTER_MAXIMUM_LENGTH,
        s.NUMERIC_PRECISION,
        s.NUMERIC_SCALE,
        s.DATETIME_PRECISION
    FROM INFORMATION
_SCHEMA.COLUMNS s
    LEFT JOIN INFORMATION_SCHEMA.COLUMNS t
        ON  t.TABLE_SCHEMA = s.TABLE_SCHEMA
        AND t.TABLE_NAME   = @tgt2
        AND t.COLUMN_NAME  = s.COLUMN_NAME
    WHERE s.TABLE_SCHEMA = @schema
      AND s.TABLE_NAME   = @src_table

      AND t.COLUMN_NAME IS NULL;

    DECLARE @sql NVARCHAR(MAX) = N'';

    SELECT @sql = @sql +
        N'ALTER TABLE ' + QUOTENAME(@schema) + N'.' + QUOTENAME(m.target_table) +
        N' ADD ' + QUOTENAME(m.column_name) + N' ' +
        CASE
        
    WHEN m.data_type IN ('varchar','nvarchar','char','nchar') THEN
                m.data_type + N'(' + CASE WHEN m.char_len = -1 THEN N'MAX' ELSE CAST(m.char_len AS NVARCHAR(20)) END + N')'
            WHEN m.data_type IN ('decimal','numeric') THEN
     
           m.data_type + N'(' + CAST(m.num_prec AS NVARCHAR(10)) + N',' + CAST(m.num_scale AS NVARCHAR(10)) + N')'
            WHEN m.data_type = 'datetime2' THEN
                m.data_type + N'(' + CAST(COALESCE(m.dt_prec, 7) AS NVARCHAR(10)) + N')'
   
         ELSE
                m.data_type
        END
        + N' NULL;'
        + CHAR(10)
    FROM @missing m;

    IF LEN(@sql) > 0
    BEGIN
        EXEC sp_executesql @sql;
    END
END
