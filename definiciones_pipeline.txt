CREATE PROCEDURE inventar_SpcHidr.sp_rep_build_targets_caso
    @caso_id INT
AS
BEGIN
    SET NOCOUNT ON;

    -- Limpia targets previos del caso
    DELETE FROM inventar_SpcHidr.rep_prod_targets
    WHERE caso_id = @caso_id;

    ;WITH base AS (
        
SELECT
            Flujo,
            COALESCE(Masa_recon, 0) AS Masa_recon,
            COALESCE(FCu, 0)        AS FCu
        FROM inventar_SpcHidr.Data_balance_rec
        WHERE caso_id = @caso_id
    ),
    srr AS (
        SELECT
            COALESCE
(SUM(Masa_recon), 0) AS masa_srr,
            COALESCE(SUM(FCu), 0)        AS fcu_srr
        FROM base
        WHERE Flujo IN ('Flujo 08','Flujo 09')
    )
    INSERT INTO inventar_SpcHidr.rep_prod_targets (caso_id, target_code, target_value, fuente)

  
  -- =========================
    -- Chancado (Flujo 03)
    -- =========================
    SELECT
        @caso_id,
        'F3',
        COALESCE(SUM(Masa_recon), 0),
        'Data_balance_rec'
    FROM base
    WHERE Flujo IN ('Flujo 03')

    UNION
 ALL

    -- =========================
    -- Sal (Flujos 10 + 11)
    -- =========================
    SELECT
        @caso_id,
        'F10F11_MASA',
        COALESCE(SUM(Masa_recon), 0),
        'Data_balance_rec'
    FROM base
    WHERE Flujo IN ('Flu
jo 10','Flujo 11')

    UNION ALL

    -- =========================
    -- Finos Cu Apilado (Flujos 14 + 15)
    -- =========================
    SELECT
        @caso_id,
        'F14F15_FCU',
        COALESCE(SUM(FCu), 0),
        'Data_balance_rec'
    
FROM base
    WHERE Flujo IN ('Flujo 14','Flujo 15')

    UNION ALL

    -- =========================
    -- Masa Apilado (Flujos 14 + 15)
    -- =========================
    SELECT
        @caso_id,
        'F14F15_MASA',
        COALESCE(SUM(Masa_recon
), 0),
        'Data_balance_rec'
    FROM base
    WHERE Flujo IN ('Flujo 14','Flujo 15')

    UNION ALL

    -- =========================
    -- Reclamo: Rotopala 1 (Flujo 19)
    -- =========================
    SELECT
        @caso_id,
        'F19__M
ASA',
        COALESCE(SUM(Masa_recon), 0),
        'Data_balance_rec'
    FROM base
    WHERE Flujo IN ('Flujo 19')

    UNION ALL

    -- =========================
    -- Reclamo: Rotopala 2 (Flujo 18)
    -- =========================
    SELECT
       
 @caso_id,
        'F18__MASA',
        COALESCE(SUM(Masa_recon), 0),
        'Data_balance_rec'
    FROM base
    WHERE Flujo IN ('Flujo 18')

    UNION ALL

    -- =========================
    -- Reclamo: Camiones Sulfuros (Flujo 21)
    -- ===========
==============
    SELECT
        @caso_id,
        'F21__MASA',
        COALESCE(SUM(Masa_recon), 0),
        'Data_balance_rec'
    FROM base
    WHERE Flujo IN ('Flujo 21')

    UNION ALL

    -- =========================
    -- Reclamo: Camiones Oxido
s (Flujo 22)
    -- =========================
    SELECT
        @caso_id,
        'F22__MASA',
        COALESCE(SUM(Masa_recon), 0),
        'Data_balance_rec'
    FROM base
    WHERE Flujo IN ('Flujo 22')

    UNION ALL

    -- =========================

    -- SRR: Masa (Flujos 08 + 09)
    -- =========================
    SELECT
        @caso_id,
        'F08F09_MASA',
        masa_srr,
        'Data_balance_rec'
    FROM srr

    UNION ALL

 08 + 09)

        @caso_id,

        fcu_srr,

    FROM srr

===============

        @caso_id,

        CASE WHEN masa_srr = 0 THEN NULL ELSE (fcu_srr / masa_srr) * 100.0 END,
        'Data_balance_rec'
    FROM srr;

END

    @mensaje NVARCHAR(2000) = NULL,
    @totales_json NVARCHAR(MAX) = NULL

        EXEC inventar_SpcHidr.sp_
rep_merge_ajustado_dyn @run_id = @run_id;

END

CREATE   PROCEDURE inventar_SpcHidr.sp_rep_conductor_pre_ajuste
    
@caso_id INT
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    DECLARE @run_id UNIQUEIDENTIFIER = NEWID();
    DECLARE @fecha_ini DATE, @fecha_fin DATE;

    -- (opcional) mantiene rep_prod_ajustado / _stg alineadas con Datos_conciliados_of
    EXE
C inventar_SpcHidr.sp_rep_prod_sync_schema;

    DECLARE @periodo TABLE (caso_id INT, fecha_inicio DATE, fecha_fin DATE);
    INSERT INTO @periodo (caso_id, fecha_inicio, fecha_fin)
    EXEC inventar_SpcHidr.sp_rep_get_periodo_caso @caso_id = @caso_id;

 
   SELECT TOP 1 @fecha_ini = fecha_inicio, @fecha_fin = fecha_fin FROM @periodo;

    IF @fecha_ini IS NULL OR @fecha_fin IS NULL
        THROW 50020, 'No se pudo obtener el período del caso (sp_rep_get_periodo_caso vacío).', 1;

    EXEC inventar_SpcHidr
.sp_rep_log_start
        @run_id = @run_id, @caso_id = @caso_id, @fecha_ini = @fecha_ini, @fecha_fin = @fecha_fin;

    -- Targets SIEMPRE desde el control oficial vigente en DEV_SIIR (ver SP anterior)
    EXEC inventar_SpcHidr.sp_rep_build_targets_caso 
@caso_id = @caso_id;

    -- Resultset 1: metadata del run
    SELECT @run_id AS run_id, @caso_id AS caso_id, @fecha_ini AS fecha_inicio, @fecha_fin AS fecha_fin;

    -- Resultset 2: targets ya construidos
    SELECT target_code, target_value
    FROM in
ventar_SpcHidr.rep_prod_targets
    WHERE caso_id = @caso_id;
END

CREATE   PROCEDURE inventar_SpcHidr.sp_rep_merge_ajustado_dyn
    @run_id UNIQUEIDENTIFIER
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @schema SYSNAME = 'inventar_SpcHidr';
    DECLARE @tgt SYSNAME = 'rep_prod_ajustado';
    DECLARE @stg SYSNAME = 'rep_prod
_ajustado_stg';

    -- columnas a excluir del UPDATE/INSERT dinámico (metadata/control)
    DECLARE @exclude TABLE (col SYSNAME);
    INSERT INTO @exclude(col) VALUES
        ('run_id'),
        ('ts_ajuste'),
        ('factor_chancado'),
        ('facto
r_sal'),
        ('factor_finos'),
        ('estado');  -- <- CLAVE: excluir estado del dinámico

    DECLARE @set NVARCHAR(MAX) = N'';
    DECLARE @ins_cols NVARCHAR(MAX) = N'';
    DECLARE @ins_vals NVARCHAR(MAX) = N'';

    ;WITH common_cols AS (
     
   SELECT c1.name AS col_name
        FROM sys.columns c1
        INNER JOIN sys.objects o1 ON o1.object_id = c1.object_id
        INNER JOIN sys.schemas s1 ON s1.schema_id = o1.schema_id
        WHERE s1.name = @schema AND o1.name = @tgt

        INTERSE
CT

        SELECT c2.name
        FROM sys.columns c2
        INNER JOIN sys.objects o2 ON o2.object_id = c2.object_id
        INNER JOIN sys.schemas s2 ON s2.schema_id = o2.schema_id
        WHERE s2.name = @schema AND o2.name = @stg
    )
    SELECT
  
      @set = @set + CASE WHEN @set = N'' THEN N'' ELSE N',' + CHAR(10) END
             + N'    tgt.' + QUOTENAME(col_name) + N' = src.' + QUOTENAME(col_name),
        @ins_cols = @ins_cols + CASE WHEN @ins_cols = N'' THEN N'' ELSE N',' END
             +
 QUOTENAME(col_name),
        @ins_vals = @ins_vals + CASE WHEN @ins_vals = N'' THEN N'' ELSE N',' END
             + N'src.' + QUOTENAME(col_name)
    FROM common_cols
    WHERE col_name NOT IN (SELECT col FROM @exclude)
      AND col_name <> 'Fecha';   
-- Fecha es clave, no se actualiza

    -- Forzar estado=1 (una sola vez)
    IF @set = N'' 
        SET @set = N'    tgt.[estado] = 1';
    ELSE
        SET @set = @set + N',' + CHAR(10) + N'    tgt.[estado] = 1';

    -- En INSERT, agregamos estado expl
ícito (una sola vez)
    SET @ins_cols = @ins_cols + CASE WHEN @ins_cols = N'' THEN N'' ELSE N',' END + N'[estado]';
    SET @ins_vals = @ins_vals + CASE WHEN @ins_vals = N'' THEN N'' ELSE N',' END + N'1';

    DECLARE @sql NVARCHAR(MAX) = N'
MERGE ' + QU
OTENAME(@schema) + N'.' + QUOTENAME(@tgt) + N' AS tgt
USING (
    SELECT *
    FROM ' + QUOTENAME(@schema) + N'.' + QUOTENAME(@stg) + N'
    WHERE run_id = @run_id
) AS src
ON tgt.[Fecha] = src.[Fecha]
WHEN MATCHED THEN
    UPDATE SET
' + @set + N'
WHEN N
OT MATCHED THEN
    INSERT (' + @ins_cols + N')
    VALUES (' + @ins_vals + N');
';

    EXEC sp_executesql @sql, N'@run_id UNIQUEIDENTIFIER', @run_id=@run_id;
END
