💻 EJEMPLO DE USO

    from modules.data_loader import get_data_loader
    from modules.reconciliation_engine import get_reconciliation_engine
    from modules.validators import get_validators
    from modules.audit_manager import get_audit_manager
    
    # Cargar datos
    recon_data = get_data_loader().get_reconciliation_results(
        'Lixiviacion_SpcHidr', run_id=5,
        fecha_inicio=date(2026, 9, 1),
        fecha_fin=date(2026, 9, 7)
    )
    
    daily_data = get_data_loader().get_daily_measured_data(
        'Lixiviacion_SpcHidr',
        fecha_inicio=date(2026, 9, 1),
        fecha_fin=date(2026, 9, 7)
    )
    
    # Procesar
    resultado = get_reconciliation_engine().process_all_flujos(
        daily_data, recon_data
    )
    datos_ajustados = resultado['datos_consolidados']
    
    # Validar
    validacion = get_validators().validate_all(
        daily_data, datos_ajustados, 200
    )
    
    # Auditoría
    audit_id = get_audit_manager().record_adjustment(
        'Lixiviacion_SpcHidr', run_id=5,
        fecha_inicio=date(2026, 9, 1),
        fecha_fin=date(2026, 9, 7),
        adjustment_detail=datos_ajustados,
        usuario='lcortes'
    )
